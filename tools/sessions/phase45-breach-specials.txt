# Phase 4.5 playtest, part 2 (headless, no player), on world-p45 after part 1 and the siege-day fix.
# Downtown is BROKEN after part 1. Checks: re-repair state by command, then a Breach on clock day 11 (game time is
# still day 1) now runs at the clock day's tier and brings special infected; the stalker hides in the dark.
until Done \(
cmd ashfall grid status
cmd forceload add -32 -112 159 79
wait 20
# Baseline: 144 force-loaded city chunks, no zombies.
cmd kill @e[type=#minecraft:zombies]

cmd spark tps
cmd time set 258000
cmd ashfall horde status
cmd ashfall grid set downtown reclaimed

cmd ashfall horde breach downtown

cmd ashfall grid status
cmd execute if entity @e[tag=ashfall_siege]
cmd execute if entity @e[type=ashfall:crawler,tag=ashfall_siege]
cmd execute if entity @e[type=ashfall:bloated,tag=ashfall_siege]
cmd execute if entity @e[type=ashfall:spitter,tag=ashfall_siege]
cmd execute if entity @e[type=ashfall:screamer,tag=ashfall_siege]
cmd execute if entity @e[type=ashfall:stalker,tag=ashfall_siege]
cmd execute if entity @e[type=ashfall:brute,tag=ashfall_siege]
# Clear the waves to see the later ones roll specials too, then hold at dawn.
cmd kill @e[tag=ashfall_siege]

cmd ashfall grid status
cmd execute if entity @e[type=#minecraft:zombies,tag=ashfall_siege,type=!minecraft:zombie]
cmd kill @e[tag=ashfall_siege]
